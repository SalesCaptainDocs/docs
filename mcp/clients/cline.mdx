<Steps>
  <Step title="Open MCP servers">
    Open the Cline panel and click the **MCP Servers** icon.
  </Step>
  <Step title="Add a remote server">
    Select **Remote Servers > Add Server**.
  </Step>
  <Step title="Enter the server details">
    Enter a name (`salescaptain`) and the server URL:

    ```text
    https://external-mcp.salescaptain.com/mcp
    ```
  </Step>
</Steps>
